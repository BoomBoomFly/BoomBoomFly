    35	 * @file circuit_breaker_params.c
    36	 *
    37	 * Circuit breaker parameters.
    38	 * Analog to real aviation circuit breakers these parameters
    39	 * allow to disable subsystems. They are not supported as standard
    40	 * operation procedure and are only provided for development purposes.
    41	 * To ensure they are not activated accidentally, the associated
    42	 * parameter needs to set to the key (magic).
    43	 */
    44	
    45	/**
    46	 * Circuit breaker for power supply check
    47	 *
    48	 * Setting this parameter to 894281 will disable the power valid
    49	 * checks in the commander.
    50	 * WARNING: ENABLING THIS CIRCUIT BREAKER IS AT OWN RISK
    51	 *
    52	 * @min 0
    53	 * @max 894281
    54	 * @category Developer
    55	 * @group Circuit Breaker
    56	 */
    57	PARAM_DEFINE_INT32(CBRK_SUPPLY_CHK, 0);
    58	
    59	/**
    60	 * Circuit breaker for IO safety
    61	 *
    62	 * Setting this parameter to 22027 will disable IO safety.
    63	 * WARNING: ENABLING THIS CIRCUIT BREAKER IS AT OWN RISK
    64	 *
    65	 * @min 0
    66	 * @max 22027
    67	 * @category Developer
    68	 * @group Circuit Breaker
    69	 */
    70	PARAM_DEFINE_INT32(CBRK_IO_SAFETY, 22027);
    71	
    72	/**
    73	 * Circuit breaker for flight termination
    74	 *
    75	 * Setting this parameter to 121212 will disable the flight termination action if triggered
    76	 * by the FailureDetector logic or if FMU is lost.
    77	 * This circuit breaker does not affect the RC loss, data link loss, geofence,
    78	 * and takeoff failure detection safety logic.
    79	 *
    80	 * @reboot_required true
    81	 * @min 0
    82	 * @max 121212
    83	 * @category Developer
    84	 * @group Circuit Breaker
    85	 */
    86	PARAM_DEFINE_INT32(CBRK_FLIGHTTERM, 121212);
    87	
    88	/**
    89	 * Circuit breaker for disabling buzzer
    90	 *
    91	 * Setting this parameter to 782097 will disable the buzzer audio notification.
    92	 *
    93	 * Setting this parameter to 782090 will disable the startup tune, while keeping
    94	 * all others enabled.
    95	 *
    96	 * @reboot_required true
    97	 * @min 0
    98	 * @max 782097
    99	 * @category Developer
   100	 * @group Circuit Breaker
   101	 */
   102	PARAM_DEFINE_INT32(CBRK_BUZZER, 0);
   103	
   104	/**
   105	 * Circuit breaker for USB link check
   106	 *
   107	 * Setting this parameter to 197848 will disable the USB connected
   108	 * checks in the commander, setting it to 0 keeps them enabled (recommended).
   109	 *
   110	 * We are generally recommending to not fly with the USB link
   111	 * connected and production vehicles should set this parameter to
   112	 * zero to prevent users from flying USB powered. However, for R&D purposes
   113	 * it has proven over the years to work just fine.
   114	 *
   115	 * @min 0
   116	 * @max 197848
   117	 * @category Developer
   118	 * @group Circuit Breaker
   119	 */
   120	PARAM_DEFINE_INT32(CBRK_USB_CHK, 197848);
   121	
   122	/**
   123	 * Circuit breaker for arming in fixed-wing mode check
   124	 *
   125	 * Setting this parameter to 159753 will enable arming in fixed-wing
   126	 * mode for VTOLs.
   127	 * WARNING: ENABLING THIS CIRCUIT BREAKER IS AT OWN RISK
   128	 *
   129	 * @min 0
   130	 * @max 159753
   131	 * @category Developer
   132	 * @group Circuit Breaker
   133	 */
   134	PARAM_DEFINE_INT32(CBRK_VTOLARMING, 0);
